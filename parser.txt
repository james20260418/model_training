    def _init_shared_state(self):
        """初始化多进程共享状态"""
        # 使用 Manager 创建进程间共享的字典
        manager = multiprocessing.Manager()
        self.shared_state = manager.dict()
        self.shared_state["first_run"] = True
        self.shared_state["count"] = 0
        self.shared_state_lock = manager.Lock()

    def get_shared_first_run(self):
        """获取共享的 first_run 状态"""
        if self.shared_state:
            with self.shared_state_lock:
                return self.shared_state["first_run"]
        return True  # 单进程模式默认返回 True

    def set_shared_first_run(self, value):
        """设置共享的 first_run 状态"""
        if self.shared_state:
            with self.shared_state_lock:
                self.shared_state["first_run"] = value

    def increment_shared_count(self):
        """原子递增共享计数器"""
        if self.shared_state:
            with self.shared_state_lock:
                self.shared_state["count"] += 1
                return self.shared_state["count"]
        return 0

    def get_storage_options_by_path(self, path: str) -> Dict:
        """根据路径前缀返回对应的存储配置"""
        for prefix, options in self._storage_options_by_prefix.items():
            if path.startswith(prefix):
                return options

        # 如果都不匹配，返回默认配置
        return self.STORAGE_OPTIONS

    def all_values_empty(self, dictionary, empty_values=None):
        values_len = 0
        for value in dictionary.values():
            values_len += len(value)
        if values_len == 0:
            return True
        else:
            return False

    def check_simple_msg(self, simple, scenario, index):
        if simple is None:
            scenario["data_check"][index]["src_label_stag_1"] = "None"
            return False
        scenario["data_check"][index]["timestamp"] = simple["timestamp"]
        scenario["data_check"][index]["src_label_stag_1"] = "Valid"
        # 检查数据中有没有想要的字段
        extract_msg = ["dat_name", "ego", "agents", "algo_env", "timestamp", "sd_map"]
        find_msg_success = True
        for msg_key in extract_msg:
            if msg_key not in simple:
                scenario["data_check"][index][msg_key] = "Not Exist"
                print("lance has not {}", msg_key)
                find_msg_success = False

        if "algo_env" in simple and not simple["algo_env"]["lanes"]:
            if not scenario["data_check"][index].get("algo_env", None):
                scenario["data_check"][index]["algo_env"] = {}
            scenario["data_check"][index]["algo_env"]["lanes"] = "Not Exist"
            find_msg_success = False

        return find_msg_success

    def check_map_valid(self, simple, scenario, index, extracted_version):
        bev_info = simple["algo_env"]
        bev_msg = ["position", "lanes", "lines", "boundaries", "junctions"]
        scenario["data_check"][index]["algo_env"] = {}
        for msg_key in bev_msg:
            if msg_key not in bev_info:
                scenario["data_check"][index]["algo_env"][msg_key] = "Not Exist"
            else:
                if (
                    bev_info[msg_key] is None
                    or (
                        isinstance(bev_info[msg_key], list)
                        or isinstance(bev_info[msg_key], dict)
                    )
                    and len(bev_info[msg_key]) == 0
                ):
                    scenario["data_check"][index]["algo_env"][msg_key] = "None"
                else:
                    # 如果是假bev数据，需要打上标签
                    if msg_key == "position":
                        scenario["data_check"][index]["algo_env_from_bev"] = False
                        position = bev_info["position"]["pose"]["position"]
                        if position["x"] != 0 or position["y"] != 0:
                            scenario["data_check"][index]["algo_env_from_bev"] = True

                    if msg_key == "junctions":
                        if (
                            "pts" in bev_info[msg_key][0]
                            and len(bev_info[msg_key][0]["pts"]) > 0
                        ):
                            scenario["data_check"][index]["algo_env"][
                                "junction_pts"
                            ] = "Exist"
                            if bev_info[msg_key][0]["pts"][0]["z"] > 0:
                                scenario["data_check"][index]["algo_env"][
                                    "junction_pts"
                                ] = "Valid"
                        if (
                            "junction_pts" in bev_info[msg_key][0]
                            and len(bev_info[msg_key][0]["junction_pts"]) > 0
                        ):
                            scenario["data_check"][index]["algo_env"][
                                "junction_pts"
                            ] = "Exist"
                            if (
                                bev_info[msg_key][0]["junction_pts"][0]["point"]["z"]
                                > 0
                                or bev_info[msg_key][0]["junction_pts"][0]["type"]
                                == "kJPTExit"
                                or bev_info[msg_key][0]["junction_pts"][0]["type"]
                                == "kJPTEntrance"
                            ):
                                scenario["data_check"][index]["algo_env"][
                                    "junction_pts"
                                ] = "Valid"
                    scenario["data_check"][index]["algo_env"][msg_key] = "Valid"

        sd_map = simple["sd_map"]

            scenario["data_check"][index]["path_info"] = "Not Exist"

                else:

                if (

                    and len(sd_map["path_info"][0]["segments"]) == 0
                ):
                    scenario["data_check"][index]["path_info"] = "None"
                else:
                    scenario["data_check"][index]["path_info"] = "Valid"

        map_data = simple["map"]
        scenario["data_check"][index]["map"] = {}
        map_msg = ["links", "car_links"]
        for msg_key in map_msg:
            if msg_key not in map_data:
                scenario["data_check"][index]["map"][msg_key] = "Not Exist"
            else:
                if (
                    map_data[msg_key] is None
                    or (

                        or isinstance(map_data[msg_key], dict)
                    )
                    and len(map_data[msg_key]) == 0
                ):
                    scenario["data_check"][index]["map"][msg_key] = "None"
                else:
                    scenario["data_check"][index]["map"][msg_key] = "Valid"

        if "route_links" not in simple:
            scenario["data_check"][index]["map"]["route_links"] = "Not Exist"
        else:
            if (
                simple["route_links"] is None

    def check_dataout_tags(self, data_out, scenario):
        scenario["pkl_valid"] = data_out["pkl_valid"]
        scenario["will_predict_path"] = data_out["Flags"]["will_predict_path"].item()
        scenario["gt_refpath_centerline_valid"] = (
            data_out["gt_refpath"]["centerline_valid_mask"].any().item()
        )
        scenario["sdmap_unmatched_gt"] = data_out["Flags"]["sdmap_unmatched_gt"].item()
        scenario["agent_predict_mask_any"] = (
            data_out["agent"]["predict_mask"].any().item()
        )
        scenario["agent_valid_mask_any"] = data_out["agent"]["valid_mask"].any().item()
        av_index = data_out.get("agent").get("av_index")
        scenario["ego_spd_kph"] = (
            np.linalg.norm(data_out["agent"]["velocity"][av_index][self.cur_frame])
            * 3.6
        )

        if "boundary_loss_valid" in data_out["Flags"]:
            scenario["boundary_loss_valid"] = data_out["Flags"]["boundary_loss_valid"]

        if self.all_values_empty(data_out["sdmap_path"]["refpaths_local"][0]):
            scenario["sdmap_path_refpaths_local"] = "Not Valid"
        else:
            scenario["sdmap_path_refpaths_local"] = "Valid"
        if data_out["cross_info"]["in_cross"]:
            scenario["in_cross"] = "Valid"
        else:
            scenario["in_cross"] = "Not Valid"

        scenario["ego_lanechange"] = data_out["Flags"]["ego_lanechange"]
        scenario["action_type"] = data_out["extra_tag"]["action_type"]
        scenario["gt_extra_tag"] = data_out["extra_tag"]["gt_extra_tag"]

        # 获取tag
        tags_msg = {
            "lanechange_tag": "lanechange_agent_id",
            "cutin_tag": "cutin_tag_agent_id",
            "merge_tag": "merge_tag_agent_id",
            "cross_tag": "cross_tag_agent_id",
        }
        scenario.setdefault("tags_msg", {})
        scenario["tags_msg"]["all_agents_id"] = data_out["agent"]["id"].tolist()
        scenario["tags_msg"]["predict_valid_agents_id"] = []
        for agent_idx, agent_predict_mask in enumerate(
            data_out["agent"]["predict_mask"]
        ):
            if agent_predict_mask.any():
                scenario["tags_msg"]["predict_valid_agents_id"].append(
                    data_out["agent"]["id"][agent_idx].item()
                )
        for tag, agent_ids in tags_msg.items():
            scenario["tags_msg"].setdefault(agent_ids, [])
            if tag in data_out["agent"]["tags"]:
                indices = np.where(data_out["agent"]["tags"][tag].numpy() == 1)
                if len(indices) > 0:
                    scenario["tags_msg"][agent_ids] = data_out["agent"]["id"][
                        indices
                    ].tolist()

        scenario["refpath_task_valid"] = data_out["task_valid"]["refpath_task_valid"]
        scenario["planning_task_valid"] = data_out["task_valid"]["planning_task_valid"]
        scenario["prediction_task_valid"] = data_out["task_valid"][
            "prediction_task_valid"
        ]
        scenario["is_nudge_scenario"] = data_out["Flags"]["is_nudge_scenario"].item()

    def load_lance(
        self,
        data_fetcher,
        dat_name,
        code_version,
        lance_stage_one_path,
        lance_sensor_path,
        frames,
        pkl_parser=True,
        lance_parser=True,
        scenario: dict = {},
        is_test_data=True,
    ) -> list:
        print(f"processing {dat_name}, valid frame number = {len(frames)}")
        images_byte = {
            "ac_front_wide_byte": bytes,
            "ac_front_narr_byte": bytes,
            "ac_back_right_byte": bytes,
            "ac_front_right_byte": bytes,
            "ac_front_left_byte": bytes,
            "ac_back_left_byte": bytes,
            "ac_back_byte": bytes,
        }

        frame_dict = {"dat_name": [], "frame_key": [], "pnc_simple": []}

        try:
            match = re.search(r"v?(\d+\.\d+\.\d+)", code_version)
            if not match:
                raise ValueError(f"无法从 {code_version} 中提取版本号")
            extracted_version = match.group(1)

            row_count = len(frames)
            sample_step = int(round(self.sample_period / 0.1))
            scenario["rows_num"] = row_count
            scenario["valid_rows_num"] = int(round(row_count / sample_step))

            all_ego_status = {}
            stoplineinfo_in_dat = {}
            sdmap_pathinfo_in_dat = {}
            if version.parse(extracted_version) < version.parse("2.4.1"):
                (
                    all_ego_status,
                    stoplineinfo_in_dat,
                    sdmap_pathinfo_in_dat,
                ) = self.extra_all_frame_info(
                    data_fetcher,
                    dat_name,
                    lance_stage_one_path,
                    frames,
                )
            lance_stage_one_infos = [(lance_stage_one_path, ["src_label_stag_1"])]

            for frame, frame_ids in list(frames.items())[::sample_step]:
                try:
                    frame_data_list = data_fetcher.fetch_frame_data(
                        dat_name, frame, lance_stage_one_infos
                    )
                except Exception as e:
                    print(
                        f"{e}, lance_infos:{lance_stage_one_infos}, dat_name: {dat_name}, frame_id: {frame}"
                    )
                simple = pickle.loads(frame_data_list[0])
                index = frame_ids["row_id"]
                scenario["data_check"][index] = {}

                if not self.check_simple_msg(simple, scenario, index):
                    continue
                k2 = frame_ids.get("sensor_row_id", -1)
                if k2 == -1:
                    continue
                if k2 > 0:
                    if index in scenario["data_check"]:
                        scenario["data_check"][index]["sensor_row_id"] = k2
                frame_key = f"{dat_name}_{frame}"

                if is_test_data:
                    lance_sensor_infos = [(lance_sensor_path, ["sensor"])]
                    try:
                        sensor_frame_list = data_fetcher.fetch_frame_data(
                            dat_name, frame, lance_sensor_infos
                        )
                    except Exception as e:
                        print(
                            f"{e}, lance_infos:{lance_sensor_infos}, dat_name: {dat_name}, frame_id: {frame}"
                        )
                    sensor_data = pickle.loads(sensor_frame_list[0])
                    images_byte["ac_front_wide_byte"] = (
                        sensor_data.get("ac_front_wide")
                        if "ac_front_wide" in sensor_data
                        else sensor_data.get("cam_front")
                    )
                    images_byte["ac_back_byte"] = (
                        sensor_data.get("ac_back")
                        if "ac_back" in sensor_data
                        else sensor_data.get("cam_rear")
                    )
                    images_byte["ac_front_left_byte"] = (
                        sensor_data.get("ac_front_left")
                        if "ac_front_left" in sensor_data
                        else sensor_data.get("cam_front_left")
                    )
                    images_byte["ac_front_right_byte"] = (
                        sensor_data.get("ac_front_right")
                        if "ac_front_right" in sensor_data
                        else sensor_data.get("cam_front_right")
                    )
                    images_byte["ac_back_left_byte"] = (
                        sensor_data.get("ac_back_left")
                        if "ac_back_left" in sensor_data
                        else sensor_data.get("cam_rear_left")
                    )
                    images_byte["ac_back_right_byte"] = (
                        sensor_data.get("ac_back_right")
                        if "ac_back_right" in sensor_data
                        else sensor_data.get("cam_rear_right")
                    )

                file_name = simple["dat_name"]
                ego_data = simple["ego"]
                agent_data = simple["agents"]
                bev_info = simple["algo_env"]
                timestamp = simple["timestamp"]
                sd_map = simple["sd_map"]
                map_data = simple["map"]

                self.check_map_valid(simple, scenario, index, extracted_version)
                if (
                    scenario["data_check"][index]["path_info"] != "Valid"
                    and len(sdmap_pathinfo_in_dat) > 0
                ):
                    closest_key = min(
                        list(sdmap_pathinfo_in_dat.keys()), key=lambda k: abs(k - index)
                    )
                    sd_map["path_info"] = sdmap_pathinfo_in_dat[closest_key]
                sd_map_fulldata = {}

                if version.parse(extracted_version) >= version.parse("2.4.1"):
                    sd_map_fulldata["path_info"] = simple["sd_map_full_data"].get(
                        "path_info", {}
                    )
                else:
                    sd_map_fulldata["path_info"] = {}
                stopline_list = []
                if len(stoplineinfo_in_dat) > 0 and index not in stoplineinfo_in_dat:
                    closest_key = min(
                        list(stoplineinfo_in_dat.keys()), key=lambda k: abs(k - index)
                    )
                    stopline_list = stoplineinfo_in_dat[closest_key]
                data_input = {}
                data_input["stopline_list"] = stopline_list
                data_input["flow_ego_trafficlight_vec"] = (
                    map_data["flow_ego_trafficlight_vectors"]
                    if "flow_ego_trafficlight_vectors" in map_data
                    else []
                )
                data_input["flow_ego_trafficlight_future_vec"] = (
                    map_data["flow_ego_trafficlight_future_vectors"]
                    if "flow_ego_trafficlight_future_vectors" in map_data
                    else []
                )
                data_input["links"] = map_data["links"]

                if self.common_cfg["algrithms_switch"]["qcnet"]:
                    try:
                        data_out = self.qcnet_parser.enum_single(
                            file_name,
                            images_byte,
                            ego_data,
                            agent_data,
                            all_ego_status,
                            bev_info,
                            sd_map,
                            sd_map_fulldata,
                            extracted_version,
                            data_input,
                            timestamp,
                            scenario,
                            save_sample=True,
                        )
                        self.check_dataout_tags(data_out, scenario["data_check"][index])
                        if lance_parser:
                            self.package_pkls(
                                data_out,
                                frame_key,
                                file_name,
                                frame_dict,
                                scenario["data_check"][index],
                            )
                        del data_out
                    except Exception as e:
                        print("qcnet enum single error: {}".format(e))
                        print(f"error file {frame_key}")
                        traceback.print_exc()

            gc.collect()  # 强制回收内存
            return True, frame_dict
        except Exception as e:
            print("lance parser failed: {}".format(e))
            traceback.print_exc()
            return False, frame_dict

    def package_pkls(self, data_out, frame_key, dat_name, frame_dict, scenario):
        if data_out is None or len(data_out) < 1 or not data_out["pkl_valid"]:
            # simple = {}
            return
        else:
            simple = data_out
        value = pickle.dumps(simple)

        scenario["pnc_row_id"] = len(frame_dict["frame_key"])

        frame_dict["dat_name"].append(dat_name)
        frame_dict["frame_key"].append(frame_key)
        frame_dict["pnc_simple"].append(value)

    def extra_all_frame_info(
        self, data_fetcher, dat_name, lance_stage_one_path, frames
    ):
        """
        提前提取所有帧的关键信息：ego状态、SDMap路径信息、停止线信息
        """
        all_ego_status = {}
        stoplineinfo_in_dat = {}
        sdmap_pathinfo_in_dat = {}

        lance_infos = [(lance_stage_one_path, ["src_label_stag_1"])]

        for frame, frame_ids in list(frames.items()):
            tmp_index = frame_ids["row_id"]
            try:
                frame_data_list = data_fetcher.fetch_frame_data(
                    dat_name, frame, lance_infos
                )
            except Exception as e:
                print(
                    f"{e}, lance_infos:{lance_infos}, dat_name: {dat_name}, frame_id: {frame}"
                )
            try:
                simple = pickle.loads(frame_data_list[0])
                if not isinstance(simple, dict):
                    continue

                # 提取基础字段
                required_keys = [
                    "dat_name",
                    "ego",
                    "agents",
                    "algo_env",
                    "timestamp",
                    "sd_map",
                ]
                if not all(k in simple for k in required_keys):
                    continue

                # 1. 提取 ego 状态
                self.get_ego_future(simple, all_ego_status)

                # 2. 提取 SDMap 路径信息
                self.get_sdmap_pathinfo(
                    simple,
                    tmp_index,
                    sdmap_pathinfo_in_dat,
                )

                # 3. 提取停止线信息
                self.get_stopline_info(simple, tmp_index, stoplineinfo_in_dat)

            except Exception as e:
                print(f"解析 frame 失败 index={tmp_index}: {e}")
                continue

        return (
            all_ego_status,
            stoplineinfo_in_dat,
            sdmap_pathinfo_in_dat,
        )

    def get_ego_future(self, simple, all_ego_status):
        """提取 ego 未来状态"""
        timestamp = simple["timestamp"]
        ego_data = simple["ego"]
        all_ego_status[timestamp] = ego_data

    def get_sdmap_pathinfo(
        self,
        simple,
        tmp_index: int,
        sdmap_pathinfo_in_dat: dict,
    ):
        """提取 SDMap 路径信息"""
        if "sd_map" in simple:
            path_info = simple["sd_map"].get("path_info")
            if path_info and self._is_valid_path_info(path_info):
                sdmap_pathinfo_in_dat[tmp_index] = path_info

    def _is_valid_path_info(self, path_info) -> bool:
        """检查 path_info 是否有效"""
        return (
            isinstance(path_info, (list, dict))
            and len(path_info) > 0
            and any(
                "segments" in item
                for item in (path_info if isinstance(path_info, list) else [path_info])
            )
        )

    def get_stopline_info(self, simple, tmp_index: int, stoplineinfo_in_dat: dict):
        """提取停止线信息"""
        ground_markings = simple.get("algo_env", {}).get("ground_markings", [])
        if not ground_markings:
            return
        ego_data = simple["ego"]
        ego_pos_global = np.array([[ego_data["pos_enu"][0], ego_data["pos_enu"][1]]])
        ego_yaw = (
            ego_data["heading_angle_global"]
            if ego_data["heading_angle_global"] < 180
            else ego_data["heading_angle_global"] - 360
        )
        ego_heading_global = np.deg2rad(ego_yaw)

        stopline_list = []
        for mark in ground_markings:
            if mark.get("ground_mark_type") != "kGMTStopLine":
                continue
            # 中心点转换
            pos_vcs = [mark["position"]["x"], mark["position"]["y"]]
            pos_enu = self.converter.ego2utm(
                [pos_vcs], ego_pos_global, ego_heading_global
            )[0]

            # 所有点转换
            pts_vcs = [[pt["x"], pt["y"]] for pt in mark.get("pts", [])]
            pts_enu = (
                self.converter.ego2utm(
                    pts_vcs, ego_pos_global, ego_heading_global
                ).tolist()
                if pts_vcs
                else []
            )

            stopline_list.append(
                {
                    "stopline_center_points": {"vcs": pos_vcs, "enu": pos_enu},
                    "stopline_points": {"vcs": pts_vcs, "enu": pts_enu},
                }
            )

        if stopline_list:
            stoplineinfo_in_dat[tmp_index] = stopline_list

    def _extract_scenario_name_from_filename(self, filename: str) -> str:
        """从文件名中提取场景名称 - 取到第一个连续数字前的所有内容"""
        # 移除扩展名
        basename = os.path.splitext(filename)[0]

        # 方案：匹配到第一个纯数字部分（由下划线分隔）
        match = re.search(r"^(.+?)(?=_\d+$|_\d+_)", basename)

        if match:
            # 找到匹配，返回场景名
            scenario_name = match.group(1)
            print(f"从文件名 '{basename}' 提取场景名: {scenario_name}")
            return scenario_name
        else:
            # 如果没有匹配，尝试匹配到第一个连续数字（不考虑下划线）
            match2 = re.search(r"^(.+?)(?=\d)", basename)
            if match2:
                scenario_name = match2.group(1).rstrip("_")  # 去掉末尾可能的下划线
                print(f"从文件名 '{basename}' 提取场景名(方案2): {scenario_name}")
                return scenario_name

        # 如果都没有匹配到，返回整个basename
        print(f"从文件名 '{basename}' 提取场景名(默认): {basename}")
        return basename

    def parse_input_from_csv(self, csv_path):

        result_dict = {}

            result_dict[dat_name] = {
                "Dat名": dat_name,
                "code_version": code_version,
                "版本号": if_version,
                "infer_type": infer_type,
                "lance_stage_one": lance_stage_one_dir,
                "lance_sensor": lance_sensor_dir,
                "lance_stage_one_label": label_path,
                "frames": {},
                "scenario": [],  # CSV 无场景信息
            }

        return result_dict

    def parse_input_from_xlsx(self, xlsx_path):
        """从 XLSX 文件解析输入数据"""
        result_dict = {}
        try:
            # 加载工作簿
            workbook = load_workbook(xlsx_path)
            sheet_names = workbook.sheetnames

            # 检查所需工作表是否存在
            if "clip输出路径" not in sheet_names:
                raise ValueError("工作表 'clip输出路径' 不存在")
            if "任务总览" not in sheet_names:
                raise ValueError("工作表 '任务总览' 不存在")
            if "运行详情" not in sheet_names:
                raise ValueError("工作表 '运行详情' 不存在")

            # 读取"任务总览"工作表，以task_id为键建立索引
            overview_sheet = workbook["任务总览"]
            overview_headers = [cell.value for cell in next(overview_sheet.iter_rows())]
            task_overview = {}

            # 确保task_id列存在
            if "task_id" not in overview_headers:
                raise ValueError("工作表 '任务总览' 中缺少 'task_id' 列")

            # 提取需要的列的索引
            required_overview_cols = ["version", "infer_type", "succeed_num"]
            col_indices = {}
            for col in required_overview_cols:
                if col in overview_headers:
                    col_indices[col] = overview_headers.index(col)

            # 遍历"任务总览"表，建立task_id索引
            for row in overview_sheet.iter_rows(min_row=2):
                row_values = [cell.value for cell in row]
                task_id = row_values[overview_headers.index("task_id")]
                if task_id and task_id not in task_overview:
                    task_overview[task_id] = {}
                    for col, idx in col_indices.items():
                        if idx < len(row_values):
                            task_overview[task_id][col] = row_values[idx]

            # 读取"运行详情"工作表，提取if_version信息
            run_detail_sheet = workbook["运行详情"]
            run_detail_headers = [
                cell.value for cell in next(run_detail_sheet.iter_rows())
            ]

            # 验证所需列是否存在
            if "task_id" not in run_detail_headers:
                raise ValueError("工作表 '运行详情' 中缺少 'task_id' 列")
            if "if_version" not in run_detail_headers:
                raise ValueError("工作表 '运行详情' 中缺少 'if_version' 列")

            # 提取if_version列的索引
            task_id_idx = run_detail_headers.index("task_id")
            if_version_idx = run_detail_headers.index("if_version")

            # 建立task_id到if_version的映射
            task_if_version = {}
            for row in run_detail_sheet.iter_rows(min_row=2):
                row_values = [cell.value for cell in row]
                if len(row_values) > if_version_idx:
                    task_id = (
                        row_values[task_id_idx]
                        if task_id_idx < len(row_values)
                        else None
                    )
                    if_version = (
                        row_values[if_version_idx]
                        if if_version_idx < len(row_values)
                        else None
                    )
                    if task_id:
                        task_if_version[task_id] = if_version

            # 读取"clip输出路径"工作表
            clip_sheet = workbook["clip输出路径"]
            clip_headers = [cell.value for cell in next(clip_sheet.iter_rows())]

            # 验证所需列是否存在
            required_clip_cols = [
                "task_id",
                "clip_name",
                "LanceSensor",
                "LanceStageOne",
            ]
            for col in required_clip_cols:
                if col not in clip_headers:
                    raise ValueError(f"工作表 'clip输出路径' 中缺少 '{col}' 列")

            # 提取需要的列的索引
            clip_col_indices = {
                col: clip_headers.index(col) for col in required_clip_cols
            }

            # 遍历每一行，关联两个工作表的数据
            for row in clip_sheet.iter_rows(min_row=2):
                row_values = [cell.value for cell in row]

                # 提取所需列的值
                task_id = row_values[clip_col_indices["task_id"]]
                clip_name = row_values[clip_col_indices["clip_name"]]
                lance_sensor = row_values[clip_col_indices["LanceSensor"]]
                lance_stage_one = row_values[clip_col_indices["LanceStageOne"]]

                # 跳过空行或无效数据
                if not task_id:
                    continue

                # 从"任务总览"中查找对应的数据
                overview_data = task_overview.get(task_id, {})

                if not overview_data:
                    # 如果在"任务总览"中找不到对应的task_id，跳过该行
                    print(
                        f"Warning: Task ID {task_id} not found in task overview sheet."
                    )
                    continue
                if overview_data.get("succeed_num") == 0:
                    # 如果该任务的succeed_num为0，跳过该行
                    print(
                        f"Warning: Task ID {task_id} has succeed_num = 0, skipping it."
                    )
                    continue

                # 获取if_version信息
                if_version = task_if_version.get(task_id, "v2.4.1")  # 默认值

                result_dict[clip_name] = {
                    "Dat名": clip_name,
                    "code_version": overview_data.get("version", "v2.4.1"),
                    "版本号": if_version,  # 添加if_version信息
                    "infer_type": overview_data.get("infer_type"),
                    "lance_stage_one": f"s3:/{lance_stage_one}",
                    "lance_sensor": f"s3:/{lance_sensor}",
                    "lance_stage_one_label": f"s3:/{lance_stage_one}/label.json",
                    "frames": {},
                    "scenario": [],  # xlsx 无场景信息
                }
        except Exception as e:
            print(f"Error processing xlsx file: {e}")
            raise e

        return result_dict

                lance_info["Dat名"],
                lance_info["code_version"],
                lance_info["lance_stage_one"],
                lance_info["lance_sensor"],
                lance_info["frames"],
                scenario=data_check_dict_local,

            bucket_name = "cn--prod--veh-algo-plan-ctrl--label"
            package_lance = PncLancePackage(bucket_name, self.save_path)

                traceback.print_exc()

    def load_and_process_input(self, input_source):
        """统一入口：加载并处理来自不同来源的输入数据"""
        input_data = {}
        is_test_data = True  # 默认值
        mode = 0

        if isinstance(input_source, (str, Path)):
            path = Path(input_source)

            if path.is_file():
                is_test_data = "test" in str(path).lower()
                if path.suffix == ".csv":
                    mode = 1
                    input_data = self.parse_input_from_csv(str(path))
                elif path.suffix == ".json":
                    mode = 2
                    input_data = self.parse_input_from_local_json(str(path))
                elif path.suffix == ".xlsx":
                    mode = 1
                    is_test_data = False
                    data = self.parse_input_from_xlsx(str(path))
                    input_data.update(data)

            elif path.is_dir():
                for root, _, files in os.walk(path):
                    for file_name in files:
                        file_path = Path(root) / file_name
                        if file_path.suffix == ".csv":
                            mode = 1
                            is_test_data = "test" in str(file_path).lower()
                            data = self.parse_input_from_csv(str(file_path))
                            input_data.update(data)
                        if file_path.suffix == ".json":
                            mode = 2
                            is_test_data = "test" in str(file_path).lower()
                            data = self.parse_input_from_local_json(str(file_path))
                            input_data.update(data)

        elif isinstance(input_source, list):
            mode = 3
            input_data = self.parse_input_from_s3_label_files(input_source)
        else:
            raise ValueError(f"不支持的 input_source 类型: {type(input_source)}")

        if self.max_workers > 1:
            self._init_shared_state()
        total = len(input_data)
        print(f"开始处理，共有{total}个DAT \n")
        save_path_org = self.qcnet_parser.save_path
        with ProcessPoolExecutor(max_workers=self.max_workers) as executor:
            futures = {
                executor.submit(
                    self.process_single_dat, data, is_test_data, save_path_org, total
                ): data
                for _, data in input_data.items()
            }
            with tqdm(total=total, desc="deal with dat", unit="dat") as pbar:
                for future in as_completed(futures):
                    dat_name, success = future.result()
                    pbar.update(1)
                    remaining = total - pbar.n
                    print(
                        f"Processing datasets (processed: {pbar.n}, remaining: {remaining}) cur dat is {dat_name}"
                    )

        if not (self.common_cfg["use_dag"] and mode in [1, 3]):
            save_path = lance_parser.save_path
            halo_record_path = f"{save_path}/halo_dat_index"
            if os.path.exists(halo_record_path):
                shutil.rmtree(halo_record_path)
            os.makedirs(halo_record_path, exist_ok=False)
            for dir_path in Path(save_path).rglob("data_check"):
                data_check_statistics_scenario(dir_path.parent, halo_record_path)
